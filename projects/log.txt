잠깐, 아까 내가 실패 조건을 설명했는데, 좋은 생각이 떠올랐어. door 를 연속 2번 지나더라도 실패하지 않도록 구조를 살짝 바꿀 수 있어. 먼저 segment 는 base_node_id 를 갖는게 아니라 base_gob_id 를 갖어. 그리고 node 와 관계 있는 gob 이 아니더라도 door gob 을 gob table 에 기록할거야. 그러면 node와 연결되지 않은 segment 도 door gob 을 기준으로 wp_point 의 위치를 계산할 수 있어. 그렇게하면 recording 은 지금 상태에서 실패하지 않게 되. 단, 그렇게 door gob 을 gob 테이블에 기록한다고 해도 wp_point 의 gob_id 는 door 가 아닌 gob 이 존재하기 때문에 fk 가 아닌 것은 동일해.
>  ### 병합 시 처리
이 부분에 대해 한 가지 언급할게 있어. root_node_id 를 재계산 한다는것은 당연히 같이 있는 rel_x 와 rel_y 를 재계산한다는 뜻이야.
gob table 에 들어가는건 node 를 생성할 때는 사용자에게 입력을 받을거야. 하지만 door gob 은 이동하면서 거쳐가는 gob 이라 자연스럽게 사용자에게 입력받거나, 혹은 계산할 수 있어. 그렇게 door gob 은 자연스럽게 확보해 두었다가 gob talble 에 insert 할거야.
그리고, 좌표를 계산할 때 root_node 의 rel_x 와 rel_y 는 0,0 이 되도록 해서 해당 graph 에 속한 node 등의 위치 계산은 root_node 의 자표와 연산 필요 없이 root_node 에 연결된 gob 과의 연산만으로 가능하도록 하면 될 것 같아. 이 때 특정 root_node 와 연결된 gob 을 빠르게 얻기 위해서 gob 을 탐색할 필요 없이 graph 테이블에 base_gob_id 를 넣으면 좋겠어.
``` sql
CREATE TABLE gob_graph (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  entry_node_id     INTEGER NOT NULL UNIQUE,
  updated_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_entry_node_id FOREIGN KEY (entry_node_id) REFERENCES gob_node(id)
);

CREATE TABLE gob_node (
  id                INTEGER NOT NULL PRIMARY KEY,
  graph_id          INTEGER NOT NULL,
  vir_x             INTEGER NOT NULL,
  vir_y             INTEGER NOT NULL,
  wp_node_id        INTEGER UNIQUE,
  resname           TEXT NOT NULL,
  updated_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_graph_id FOREIGN KEY (graph_id) REFERENCES gob_graph(id),
  CONSTRAINT fk_wp_node_id FOREIGN KEY (wp_node_id) REFERENCES wp_node(id)
);
CREATE INDEX IF NOT EXISTS idx_gob_node_graph_id ON gob_node (graph_id);

CREATE TABLE gob_edge (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  node0_id          INTEGER NOT NULL,
  node1_id          INTEGER NOT NULL,
  updated_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_node0_id FOREIGN KEY (node0_id) REFERENCES gob_node(id),
  CONSTRAINT fk_node1_id FOREIGN KEY (node1_id) REFERENCES gob_node(id),
  UNIQUE (node0_id, node1_id)
);

CREATE TABLE wp_node (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  gob_graph_id      INTEGER NOT NULL,
  gob_node_id       INTEGER NOT NULL UNIQUE,
  vir_x             INTEGER NOT NULL,
  vir_y             INTEGER NOT NULL,
  name              TEXT NOT NULL,
  updated_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_gob_graph_id FOREIGN KEY (gob_graph_id) REFERENCES gob_graph(id),
  CONSTRAINT fk_gob_node_id FOREIGN KEY (gob_node_id) REFERENCES gob_node(id)
);
CREATE INDEX IF NOT EXISTS idx_wp_node_gob_graph_id ON wp_node (gob_graph_id);

CREATE TABLE wp_edge (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  node0_id          INTEGER NOT NULL,
  node1_id          INTEGER NOT NULL,
  direction         INTEGER NOT NULL,
  time_cost         DOUBLE NOT NULL,
  fatigue_cost      DOUBLE NOT NULL,
  updated_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_node0_id FOREIGN KEY (node0_id) REFERENCES wp_node(id),
  CONSTRAINT fk_node1_id FOREIGN KEY (node1_id) REFERENCES wp_node(id),
  CONSTRAINT chk_direction CHECK (direction IN (0, 1, 2, 3)),
  UNIQUE (node0_id, node1_id)
);

CREATE TABLE wp_segment (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  gob_graph_id      INTEGER NOT NULL,
  edge_id           INTEGER NOT NULL,
  step              INTEGER NOT NULL,
  updated_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_gob_graph_id FOREIGN KEY (gob_graph_id) REFERENCES gob_graph(id),
  CONSTRAINT fk_edge_id FOREIGN KEY (edge_id) REFERENCES wp_edge(id),
  UNIQUE (edge_id, step)
);
CREATE INDEX IF NOT EXISTS idx_wp_segment_gob_graph_id ON wp_segment (gob_graph_id)

CREATE TABLE wp_point (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  segment_id        INTEGER NOT NULL,
  step              INTEGER NOT NULL,
  vir_x             INTEGER NOT NULL,
  vir_y             INTEGER NOT NULL,
  mouse_button      INTEGER NOT NULL,
  gob_id            INTEGER,
  mesh_id           INTEGER,
  updated_at        DATETIME DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT fk_segment_id FOREIGN KEY (segment_id) REFERENCES wp_segment(id),
  UNIQUE (segment_id, step)
);
```
